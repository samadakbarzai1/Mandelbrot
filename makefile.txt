# Compiler
CXX = g++
CXXFLAGS = -std=c++11 -Wall -pthread

# SFML library paths 
LIBS = -lsfml-graphics -lsfml-window -lsfml-system

# Executable name
TARGET = mandelbrot

# Source files
SRCS = main.cpp ComplexPlane.cpp

# Object files
OBJS = $(SRCS:.cpp=.o)

# Default target
all: $(TARGET)

$(TARGET): $(OBJS)
	$(CXX) $(CXXFLAGS) -o $@ $^ $(LIBS)

%.o: %.cpp ComplexPlane.h
	$(CXX) $(CXXFLAGS) -c $<

clean:
	rm -f $(OBJS) $(TARGET)
